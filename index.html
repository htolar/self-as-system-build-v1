Five Canvas Sketches

Sketch 1 — Irregular Ovals

The rule/ mechanism being tested is the sigil/ the ability to make random shapes/ the same style shape. At first I tested two different things so I had to remove that and it was too jagged for my taste originally so I got rid of that too. Finally for this one there were no input controls.

JavaScript drawing code
const canvas = document.getElementById("canvas");
const ctx = canvas.getContext("2d");

let W, H;

function resize() {
    W = canvas.width = innerWidth;
    H = canvas.height = innerHeight;

    createBlobs();
}

window.addEventListener("resize", resize);

const ovalWidth = 90;
const ovalHeight = 45;

const spacingX = 120;
const spacingY = 75;

let blobs = [];

function createBlobs() {
    blobs = [];
    const columns = Math.ceil(W / spacingX) + 1;
    const rows = Math.ceil(H / spacingY) + 1;

    for (let row = 0; row < rows; row++) {
        for (let col = 0; col < columns; col++) {
            blobs.push({
                x: col * spacingX + spacingX / 2,
                y: row * spacingY + spacingY / 2,
                width: ovalWidth * (0.85 + Math.random() * 0.3),
                height: ovalHeight * (0.85 + Math.random() * 0.3),
                rotation: (Math.random() - 0.5) * 0.2,
                points: 40,
                shape: Array.from({ length: 40 }, () => 0.94 + Math.random() * 0.12)
            });
        }
    }
}

function drawBlob(blob) {
    ctx.save();
    ctx.translate(blob.x, blob.y);
    ctx.rotate(blob.rotation);
    ctx.beginPath();

    for (let i = 0; i <= blob.points; i++) {
        const index = i % blob.points;
        const angle = (index / blob.points) * Math.PI * 2;
        const x = Math.cos(angle) * (blob.width / 2) * blob.shape[index];
        const y = Math.sin(angle) * (blob.height / 2) * blob.shape[index];

        if (i === 0) ctx.moveTo(x, y);
        else ctx.lineTo(x, y);
    }

    ctx.closePath();
    ctx.fillStyle = "rgba(100, 220, 210, 0.25)";
    ctx.fill();
    ctx.strokeStyle = "rgba(170, 255, 245, 0.5)";
    ctx.lineWidth = 1.5;
    ctx.stroke();
    ctx.restore();
}

function draw() {
    ctx.clearRect(0, 0, W, H);
    for (const blob of blobs) drawBlob(blob);
    requestAnimationFrame(draw);
}

resize();
draw();

Sketch 2 — Mouse-Deformed Ovals

For this the mechanism being tested is the deforming when the cursor is touching it. The input for this one is you dragging around your mouse/ cursor which in turn outputs the disformation of the oval. It also still made it too jagged originally so I asked it to make it more smooth.

JavaScript drawing code
const canvas = document.getElementById("canvas");
const ctx = canvas.getContext("2d");

let W, H;

function resize() {
    W = canvas.width = innerWidth;
    H = canvas.height = innerHeight;
    createBlobs();
}

window.addEventListener("resize", resize);

const mouse = { x: -1000, y: -1000 };

window.addEventListener("mousemove", e => {
    mouse.x = e.clientX;
    mouse.y = e.clientY;
});

const ovalWidth = 85;
const ovalHeight = 45;
const spacingX = 120;
const spacingY = 75;

let blobs = [];

function createBlobs() {
    blobs = [];
    const columns = Math.ceil(W / spacingX) + 1;
    const rows = Math.ceil(H / spacingY) + 1;

    for (let row = 0; row < rows; row++) {
        for (let col = 0; col < columns; col++) {
            blobs.push({
                x: col * spacingX + spacingX / 2,
                y: row * spacingY + spacingY / 2,
                width: ovalWidth * (0.85 + Math.random() * 0.3),
                height: ovalHeight * (0.85 + Math.random() * 0.3),
                rotation: (Math.random() - 0.5) * 0.15,
                points: 40,
                deformation: Array(40).fill(0)
            });
        }
    }
}

function updateBlobs() {
    for (const blob of blobs) {
        for (let i = 0; i < blob.points; i++) {
            const angle = (i / blob.points) * Math.PI * 2;
            const baseX = blob.x + Math.cos(angle) * blob.width / 2;
            const baseY = blob.y + Math.sin(angle) * blob.height / 2;
            const dx = mouse.x - baseX;
            const dy = mouse.y - baseY;
            const distance = Math.sqrt(dx * dx + dy * dy);

            if (distance < 35) {
                const strength = 1 - distance / 35;
                blob.deformation[i] += strength * 8;
            }
            blob.deformation[i] *= 0.88;
        }
    }
}

function drawBlob(blob) {
    ctx.save();
    ctx.translate(blob.x, blob.y);
    ctx.rotate(blob.rotation);
    ctx.beginPath();

    for (let i = 0; i <= blob.points; i++) {
        const index = i % blob.points;
        const angle = (index / blob.points) * Math.PI * 2;
        const cos = Math.cos(angle);
        const sin = Math.sin(angle);

        let x = cos * blob.width / 2;
        let y = sin * blob.height / 2;
        const deformation = blob.deformation[index];

        x += cos * deformation;
        y += sin * deformation;

        if (i === 0) ctx.moveTo(x, y);
        else ctx.lineTo(x, y);
    }

    ctx.closePath();
    ctx.fillStyle = "rgba(100, 220, 210, 0.25)";
    ctx.fill();
    ctx.strokeStyle = "rgba(170, 255, 245, 0.5)";
    ctx.lineWidth = 1.5;
    ctx.stroke();
    ctx.restore();
}

function draw() {
    ctx.clearRect(0, 0, W, H);
    updateBlobs();
    for (const blob of blobs) drawBlob(blob);
    requestAnimationFrame(draw);
}

resize();
draw();

Sketch 3 — Draggable Water Blobs

The mechanism being tested is the dragging of the circles and seeing how they are effected. The input is clicking and dragging with the cursor. I didn't adjust anything when done.

JavaScript drawing code
const canvas = document.getElementById("canvas");
const ctx = canvas.getContext("2d");

let W, H;
const ovalWidth = 90;
const ovalHeight = 45;
const spacingX = 120;
const spacingY = 75;

let blobs = [];

const mouse = { x: 0, y: 0, down: false };
let draggedBlob = null;
let grabOffsetX = 0;
let grabOffsetY = 0;

canvas.addEventListener("pointerdown", e => {
    mouse.x = e.clientX;
    mouse.y = e.clientY;
    mouse.down = true;

    for (let i = blobs.length - 1; i >= 0; i--) {
        const blob = blobs[i];
        const dx = mouse.x - blob.x;
        const dy = mouse.y - blob.y;
        const inside = (dx * dx) / ((blob.width / 2) ** 2) + (dy * dy) / ((blob.height / 2) ** 2) < 1;

        if (inside) {
            draggedBlob = blob;
            grabOffsetX = mouse.x - blob.x;
            grabOffsetY = mouse.y - blob.y;
            blobs.splice(i, 1);
            blobs.push(blob);
            break;
        }
    }
});

canvas.addEventListener("pointermove", e => {
    mouse.x = e.clientX;
    mouse.y = e.clientY;
    if (draggedBlob) {
        draggedBlob.targetX = mouse.x - grabOffsetX;
        draggedBlob.targetY = mouse.y - grabOffsetY;
    }
});

window.addEventListener("pointerup", () => {
    mouse.down = false;
    if (draggedBlob) {
        draggedBlob.dragging = false;
        draggedBlob = null;
    }
});

function createBlobs() {
    blobs = [];
    const columns = Math.ceil(W / spacingX) + 1;
    const rows = Math.ceil(H / spacingY) + 1;

    for (let row = 0; row < rows; row++) {
        for (let col = 0; col < columns; col++) {
            const x = col * spacingX + spacingX / 2;
            const y = row * spacingY + spacingY / 2;

            blobs.push({
                x: x, y: y, targetX: x, targetY: y,
                vx: 0, vy: 0,
                width: ovalWidth * (0.85 + Math.random() * 0.3),
                height: ovalHeight * (0.85 + Math.random() * 0.3),
                rotation: (Math.random() - 0.5) * 0.2,
                points: 40,
                shape: Array.from({ length: 40 }, () => 0.94 + Math.random() * 0.12),
                deformation: Array(40).fill(0),
                deformationVelocity: Array(40).fill(0)
            });
        }
    }
}

function updateBlobs() {
    for (const blob of blobs) {
        if (blob === draggedBlob) {
            blob.targetX = mouse.x - grabOffsetX;
            blob.targetY = mouse.y - grabOffsetY;
        }

        const dx = blob.targetX - blob.x;
        const dy = blob.targetY - blob.y;

        blob.vx += dx * 0.08;
        blob.vy += dy * 0.08;
        blob.vx *= 0.78;
        blob.vy *= 0.78;
        blob.x += blob.vx;
        blob.y += blob.vy;

        for (let i = 0; i < blob.points; i++) {
            const angle = (i / blob.points) * Math.PI * 2;
            const movement = blob.vx * Math.cos(angle) + blob.vy * Math.sin(angle);

            blob.deformationVelocity[i] += -movement * 0.35;
            blob.deformationVelocity[i] += -blob.deformation[i] * 0.08;
            blob.deformationVelocity[i] *= 0.86;
            blob.deformation[i] += blob.deformationVelocity[i];
            blob.deformation[i] *= 0.995;
        }
    }
}

function drawBlob(blob) {
    ctx.save();
    ctx.translate(blob.x, blob.y);
    ctx.rotate(blob.rotation);
    ctx.beginPath();

    for (let i = 0; i <= blob.points; i++) {
        const index = i % blob.points;
        const angle = (index / blob.points) * Math.PI * 2;
        const deformation = blob.deformation[index];
        const radius = blob.shape[index];

        const x = Math.cos(angle) * (blob.width / 2) * radius + Math.cos(angle) * deformation;
        const y = Math.sin(angle) * (blob.height / 2) * radius + Math.sin(angle) * deformation;

        if (i === 0) ctx.moveTo(x, y);
        else ctx.lineTo(x, y);
    }

    ctx.closePath();
    ctx.fillStyle = "rgba(100, 220, 210, 0.25)";
    ctx.fill();
    ctx.strokeStyle = "rgba(170, 255, 245, 0.5)";
    ctx.lineWidth = 1.5;
    ctx.stroke();
    ctx.restore();
}

function draw() {
    ctx.clearRect(0, 0, W, H);
    updateBlobs();
    for (const blob of blobs) drawBlob(blob);
    requestAnimationFrame(draw);
}

function resize() {
    W = canvas.width = window.innerWidth;
    H = canvas.height = window.innerHeight;
    createBlobs();
}

window.addEventListener("resize", resize);
resize();
draw();

Sketch 4 — Gravity + Water Blobs

The rule being tested in this final one is gravity and how they affect the looks and behavior of the blob. Their is no input for this test, and there were no changes made for this one.

JavaScript drawing code
const canvas = document.getElementById("canvas");
const ctx = canvas.getContext("2d");

let W, H;
const ovalWidth = 90;
const ovalHeight = 45;
const spacingX = 120;
const spacingY = 75;
const gravity = 0.18;
const bounce = 0.55;

let blobs = [];
const mouse = { x: 0, y: 0 };
let draggedBlob = null;
let grabOffsetX = 0;
let grabOffsetY = 0;

canvas.addEventListener("pointerdown", e => {
    mouse.x = e.clientX;
    mouse.y = e.clientY;

    for (let i = blobs.length - 1; i >= 0; i--) {
        const blob = blobs[i];
        const dx = mouse.x - blob.x;
        const dy = mouse.y - blob.y;
        const inside = (dx * dx) / ((blob.width / 2) ** 2) + (dy * dy) / ((blob.height / 2) ** 2) < 1;

        if (inside) {
            draggedBlob = blob;
            grabOffsetX = mouse.x - blob.x;
            grabOffsetY = mouse.y - blob.y;
            blobs.splice(i, 1);
            blobs.push(blob);
            break;
        }
    }
});

canvas.addEventListener("pointermove", e => {
    mouse.x = e.clientX;
    mouse.y = e.clientY;
});

window.addEventListener("pointerup", () => {
    if (draggedBlob) {
        draggedBlob.vx *= 1.2;
        draggedBlob.vy *= 1.2;
        draggedBlob = null;
    }
});

function createBlobs() {
    blobs = [];
    const columns = Math.ceil(W / spacingX) + 1;
    const rows = Math.ceil(H / spacingY) + 1;

    for (let row = 0; row < rows; row++) {
        for (let col = 0; col < columns; col++) {
            const x = col * spacingX + spacingX / 2;
            const y = row * spacingY + spacingY / 2;

            blobs.push({
                x: x, y: y,
                vx: 0, vy: 0,
                width: ovalWidth * (0.85 + Math.random() * 0.3),
                height: ovalHeight * (0.85 + Math.random() * 0.3),
                rotation: (Math.random() - 0.5) * 0.2,
                points: 40,
                shape: Array.from({ length: 40 }, () => 0.94 + Math.random() * 0.12),
                deformation: Array(40).fill(0),
                deformationVelocity: Array(40).fill(0)
            });
        }
    }
}

function updateBlobs() {
    for (const blob of blobs) {
        if (blob === draggedBlob) {
            blob.x = mouse.x - grabOffsetX;
            blob.y = mouse.y - grabOffsetY;
            blob.vx *= 0.8;
            blob.vy *= 0.8;
        } else {
            blob.vy += gravity;
            blob.vx *= 0.995;
            blob.vy *= 0.995;
            blob.x += blob.vx;
            blob.y += blob.vy;

            const bottom = H - blob.height / 2;
            if (blob.y > bottom) {
                blob.y = bottom;
                blob.vy *= -bounce;
                blob.vx *= 0.88;
                if (Math.abs(blob.vy) < 0.5) blob.vy = 0;
            }

            const left = blob.width / 2;
            const right = W - blob.width / 2;

            if (blob.x < left) { blob.x = left; blob.vx *= -bounce; }
            if (blob.x > right) { blob.x = right; blob.vx *= -bounce; }
        }

        for (let i = 0; i < blob.points; i++) {
            const angle = (i / blob.points) * Math.PI * 2;
            const movement = blob.vx * Math.cos(angle) + blob.vy * Math.sin(angle);

            blob.deformationVelocity[i] -= movement * 0.25;
            blob.deformationVelocity[i] -= blob.deformation[i] * 0.08;
            blob.deformationVelocity[i] *= 0.86;
            blob.deformation[i] += blob.deformationVelocity[i];
        }
    }
}

function drawBlob(blob) {
    ctx.save();
    ctx.translate(blob.x, blob.y);
    ctx.rotate(blob.rotation);
    ctx.beginPath();

    for (let i = 0; i <= blob.points; i++) {
        const index = i % blob.points;
        const angle = (index / blob.points) * Math.PI * 2;
        const deformation = blob.deformation[index];
        const radius = blob.shape[index];

        const x = Math.cos(angle) * (blob.width / 2) * radius + Math.cos(angle) * deformation;
        const y = Math.sin(angle) * (blob.height / 2) * radius + Math.sin(angle) * deformation;

        if (i === 0) ctx.moveTo(x, y);
        else ctx.lineTo(x, y);
    }

    ctx.closePath();
    ctx.fillStyle = "rgba(100, 220, 210, 0.25)";
    ctx.fill();
    ctx.strokeStyle = "rgba(170, 255, 245, 0.5)";
    ctx.lineWidth = 1.5;
    ctx.stroke();
    ctx.restore();
}

function draw() {
    ctx.clearRect(0, 0, W, H);
    updateBlobs();
    for (const blob of blobs) drawBlob(blob);
    requestAnimationFrame(draw);
}

function resize() {
    W = canvas.width = window.innerWidth;
    H = canvas.height = window.innerHeight;
    createBlobs();
}

window.addEventListener("resize", resize);
resize();
draw();
